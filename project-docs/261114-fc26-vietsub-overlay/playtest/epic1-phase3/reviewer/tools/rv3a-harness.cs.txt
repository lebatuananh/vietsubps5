// ===================== helpers =====================
async Task Flicker(bool animateItemCell)
{
    var cfg = new EngineConfig();
    var home = Tmp();
    var svc = new TranslationService(glossary, TranslationCache.Load(null, Path.Combine(home, "c.json")), null,
        new UsageCounter(Path.Combine(home, "u"), 100000, () => new DateOnly(2026, 11, 18)), () => now, cfg.Tuning);
    var sink = new RecSink();
    var ocr = new FakeOcr([Line("Board", 20, 20, 90, 28)]);
    var pipe = new FramePipeline(cfg, ocr, svc, sink, "s");
    var basePx = Solid(1920, 1080, 30).Bgra;
    int transitions = 0; bool prev = false; long lastSendWithItem = -1, maxGap = 0, ocrCalls0 = 0;
    for (long t = 0; t <= 10000; t += 250)
    {
        var px = (byte[])basePx.Clone();
        byte v = (byte)((t / 250) % 2 == 0 ? 60 : 110);
        int cx0 = animateItemCell ? 0 : 1800, cy0 = animateItemCell ? 0 : 960;
        for (int y = cy0; y < cy0 + 120; y++) for (int x = cx0; x < cx0 + 120; x++) { int i = (y * 1920 + x) * 4; if (animateItemCell && x >= 20 && x < 110 && y >= 20 && y < 48) continue; px[i] = px[i + 1] = px[i + 2] = v; }
        int before = sink.Count;
        await pipe.OnFrameAsync(new SourceFrame(px, 1920, 1080, t, null), default);
        bool has = sink.LastItems > 0;
        if (sink.Count > before)
        {
            if (has != prev) transitions++;
            prev = has;
            if (has) { if (lastSendWithItem >= 0) maxGap = Math.Max(maxGap, t - lastSendWithItem); lastSendWithItem = t; }
        }
        else if (prev && lastSendWithItem >= 0) maxGap = Math.Max(maxGap, t - lastSendWithItem);
    }
    Console.WriteLine($"  gói gửi={sink.Count}, OCR={ocr.Calls}, số lần item hiện↔mất={transitions}, khe dài nhất không có gói khi item đang hiện={maxGap} ms (ttlMs={cfg.Tuning.TtlMs}), trạng thái cuối={pipe.State}");
}

static ClaudeRawResponse OkAll(ClaudeRequest r)
{
    using var doc = JsonDocument.Parse(r.User);
    var items = doc.RootElement.EnumerateArray().Select(e => new { id = e.GetProperty("id").GetString(), vi = "VI " + e.GetProperty("en").GetString() });
    return new ClaudeRawResponse(JsonSerializer.Serialize(new { items }), "end_turn", 1, 1, 0, 0);
}
static SourceFrame Solid(int w, int h, byte v) { var p = new byte[w * h * 4]; for (int i = 0; i < p.Length; i += 4) { p[i] = p[i + 1] = p[i + 2] = v; p[i + 3] = 255; } return new SourceFrame(p, w, h, 0, null); }
static OcrLineData Line(string text, double x, double y, double w, double h)
{
    var parts = text.Split(' '); double cw = w / Math.Max(1, text.Length); double cx = x; var words = new List<OcrWordData>();
    foreach (var p in parts) { words.Add(new OcrWordData(p, new Box(cx, y, p.Length * cw, h))); cx += (p.Length + 1) * cw; }
    return new OcrLineData(text, new Box(x, y, w, h), words);
}

sealed class CountingTransport(Func<ClaudeRequest, ClaudeRawResponse> f) : IClaudeTransport
{
    public int Calls;
    public Task<ClaudeRawResponse> SendAsync(ClaudeRequest r, CancellationToken ct) { Interlocked.Increment(ref Calls); return Task.FromResult(f(r)); }
}
sealed class FakeOcr(IReadOnlyList<OcrLineData> lines) : IOcrProvider
{
    public int Calls;
    public Task<IReadOnlyList<OcrLineData>> RecognizeAsync(SourceFrame frame, CancellationToken ct) { Calls++; return Task.FromResult(lines); }
}
sealed class RecSink : IFrameSink
{
    public int Count, LastItems, LastBytes; public long LastAt = -1;
    public void Send(string line)
    {
        Count++; LastBytes = System.Text.Encoding.UTF8.GetByteCount(line);
        var f = ProtocolJson.ParseFrame(line)!; LastItems = f.Items.Count;
    }
}
